rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function realUser() { return request.auth != null && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == 'a2224907@gmail.com'
        && request.auth.token.email_verified == true;
    }

    // ---- Memorama 翻牌記憶 排行榜 ----
    match /memory_game_leaderboard/{docId} {
      function m() { return request.resource.data; }
      function okMemorama() {
        return m().keys().hasOnly(['name','score','moves','timeSeconds','script','lang','createdAt'])
          && m().keys().hasAll(['name','score','moves','timeSeconds','createdAt'])
          && m().name is string && m().name.size() >= 1 && m().name.size() <= 14
          && m().moves is int && m().moves >= 18 && m().moves <= 2000
          && m().timeSeconds is int && m().timeSeconds >= 5
          && m().score is int && m().score >= 0 && m().score <= 1000
          // 分數必須跟次數、時間對得上（遊戲的計分公式）
          && (
            (m().timeSeconds < 300 && m().score <= 12600 / m().moves + 301 - m().timeSeconds)
            || (m().timeSeconds >= 300 && m().score <= 12600 / m().moves + 1)
          )
          && m().createdAt == request.time;
      }
      allow read: if true;
      allow create: if signedIn() && okMemorama();
      allow update: if false;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }

    // ---- Memorama 連線房間（兩人對戰／一起玩；房間 6 小時後過期） ----
    match /memorama_rooms/{code} {
      function r() { return request.resource.data; }
      function old() { return resource.data; }
      function okName(n) { return n is string && n.size() >= 1 && n.size() <= 14; }
      // 已過期或已關閉的房間，任何人都可以重新開一間同代碼的新房間，或把它刪掉
      function reclaimable() { return old().expireAt < request.time || old().status == 'closed'; }
      function isMember() {
        return request.auth.uid == old().p0.uid
          || (old().p1 != null && request.auth.uid == old().p1.uid);
      }
      function okShape() {
        return r().keys().hasOnly(['code','mode','hostUid','p0','p1','seen0','seen1','status','round',
                                   'deck','matched','open','openBy','turn','scores','moves','flipSeq',
                                   'lastFlip','startedAt','finishedAt','createdAt','expireAt'])
          && r().status in ['waiting','playing','finished','closed']
          && r().deck is list && r().deck.size() <= 36
          && r().matched is list && r().matched.size() <= 36
          && r().open is list && r().open.size() <= 2
          && r().expireAt is timestamp
          && r().expireAt <= request.time + duration.value(7, 'h');
      }
      function validNew() {
        return code.matches('^[0-9]{4}$')
          && r().code == code
          && r().mode in ['vs','coop']
          && r().hostUid == request.auth.uid
          && r().p0.uid == request.auth.uid && okName(r().p0.name)
          && r().p1 == null
          && r().status == 'waiting'
          && okShape();
      }
      // 房主、房間代碼、模式建立後就不能改
      function sameCore() {
        return r().code == old().code && r().mode == old().mode
          && r().hostUid == old().hostUid && r().p0 == old().p0;
      }
      // 第二個人加入：只能填自己的名字，其他欄位都不能動
      function joining() {
        return old().p1 == null && old().status == 'waiting'
          && r().p1.uid == request.auth.uid && okName(r().p1.name)
          && r().diff(old()).affectedKeys().hasOnly(['p1','seen1']);
      }
      // 房間裡的兩個人：可以翻牌、開新局；第二個人只能「維持不變」或「離開」
      function memberUpdate() {
        return isMember() && sameCore() && okShape()
          && (r().p1 == old().p1 || r().p1 == null);
      }
      allow read: if signedIn();
      allow create: if signedIn() && validNew();
      allow update: if signedIn() && (
        (reclaimable() && validNew())
        || (!reclaimable() && (joining() || memberUpdate()))
      );
      allow delete: if signedIn() && (reclaimable() || request.auth.uid == old().hostUid);
    }

    // ---- Ordena el Armario 整理衣櫃 排行榜（每台裝置一筆，只留最好成績） ----
    match /ordena_leaderboard/{uid} {
      function o() { return request.resource.data; }
      function okOrdena() {
        return o().keys().hasOnly(['name','errors','secs','rank','textAid','script','lang','updatedAt'])
          && o().keys().hasAll(['name','errors','secs','rank','textAid','updatedAt'])
          && o().name is string && o().name.size() >= 1 && o().name.size() <= 14
          && o().errors is int && o().errors >= 0 && o().errors <= 999
          && o().secs is int && o().secs >= 10 && o().secs <= 99999
          && o().textAid is bool
          // 排名：錯誤越少越前面，錯誤一樣比時間
          && o().rank is int
          && o().rank == (999 - o().errors) * 100000 + (99999 - o().secs)
          && o().updatedAt == request.time;
      }
      allow read: if true;
      allow create: if signedIn() && request.auth.uid == uid && okOrdena();
      allow update: if signedIn() && request.auth.uid == uid && okOrdena()
                    && o().rank >= resource.data.rank;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }

    // ---- 色數消消樂 排行榜 ----
    match /colorgame_boards/{board}/entries/{uid} {
      function validBoard() { return board.matches('^[AB]_(20|30|42)_(basic|adv)_(999|9999)$'); }
      function maxScore() {
        return board.matches('^[AB]_20_.*') ? 1150 : (board.matches('^[AB]_30_.*') ? 1650 : 2250);
      }
      function d() { return request.resource.data; }
      function okData() {
        return d().keys().hasOnly(['name','score','secs','hearts','cleared','rank','updatedAt'])
          && d().keys().hasAll(['name','score','secs','hearts','cleared','rank','updatedAt'])
          && d().name is string && d().name.size() >= 1 && d().name.size() <= 24
          && d().score is int && d().score >= 0 && d().score <= maxScore()
          && d().secs is int && d().secs >= 0 && d().secs <= 99999
          && d().hearts is int && d().hearts >= 0 && d().hearts <= 5
          && d().cleared is bool
          && d().rank is int
          && d().rank == d().score * 100000 + (99999 - d().secs)
          && d().updatedAt == request.time;
      }
      allow read: if signedIn() && validBoard();
      allow create: if signedIn() && request.auth.uid == uid && validBoard() && okData();
      allow update: if signedIn() && request.auth.uid == uid && validBoard() && okData()
                    && d().rank >= resource.data.rank;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }


    // ---- 井遊戲：誰還在井裡（每個人只有自己的一筆；按 Y 離開就刪掉） ----
    match /well_visitors/{uid} {
      function w() { return request.resource.data; }
      function okWell() {
        return w().keys().hasOnly(['enteredAt','lastSeen'])
          && w().keys().hasAll(['enteredAt','lastSeen'])
          && w().enteredAt is timestamp
          && w().lastSeen == request.time;
      }
      allow read: if true;
      allow create, update: if signedIn() && request.auth.uid == uid && okWell();
      allow delete: if signedIn() && request.auth.uid == uid;
    }

    // ---- 文字拼圖 排行榜（依片數與難度分榜，每台裝置每榜只留最好成績） ----
    // 榜名：'8'、'10'、'12'、'15' = 輕鬆；'8r'、'10r'、'12r'、'15r' = 進階（會旋轉）
    match /jigsaw_boards/{board}/entries/{uid} {
      function validBoard() { return board in ['8','10','12','15','8r','10r','12r','15r']; }
      function j() { return request.resource.data; }
      function okJigsaw() {
        return j().keys().hasOnly(['name','secs','moves','text','rank','script','lang','updatedAt'])
          && j().keys().hasAll(['name','secs','moves','text','rank','updatedAt'])
          && j().name is string && j().name.size() >= 1 && j().name.size() <= 14
          && j().text is string && j().text.size() >= 1 && j().text.size() <= 80
          && j().secs is int && j().secs >= 30 && j().secs <= 99999
          && j().moves is int && j().moves >= 64 && j().moves <= 9999
          && j().rank is int
          && j().rank == (99999 - j().secs) * 10000 + (9999 - j().moves)
          && j().updatedAt == request.time;
      }
      allow read: if signedIn() && validBoard();
      allow create: if signedIn() && request.auth.uid == uid && validBoard() && okJigsaw();
      allow update: if signedIn() && request.auth.uid == uid && validBoard() && okJigsaw()
                    && j().rank >= resource.data.rank;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }
    // ================= 以下 5 個遊戲的公開排行榜（每台裝置每個榜只留一筆最好成績） =================

    // ---- 幫媽媽買東西 ----
    match /mama_shopping_leaderboard/{uid} {
      function s() { return request.resource.data; }
      function okMama() {
        return s().keys().hasOnly(['name','score','updatedAt'])
          && s().name is string && s().name.size() >= 1 && s().name.size() <= 14
          && s().score is int && s().score >= 0 && s().score <= 100
          && s().updatedAt == request.time;
      }
      allow read: if true;
      allow create: if signedIn() && request.auth.uid == uid && okMama();
      allow update: if signedIn() && request.auth.uid == uid && okMama()
                    && s().score >= resource.data.score;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }

    // ---- 把它放好（3 個關卡各記最高分，總分 = 三關相加） ----
    match /baLeaderboard/{uid} {
      function b() { return request.resource.data; }
      function okBa() {
        return b().keys().hasOnly(['name','l1','l2','l3','total','updatedAt'])
          && b().name is string && b().name.size() >= 1 && b().name.size() <= 14
          && b().l1 is int && b().l1 >= 0 && b().l1 <= 5000
          && b().l2 is int && b().l2 >= 0 && b().l2 <= 5000
          && b().l3 is int && b().l3 >= 0 && b().l3 <= 5000
          && b().total == b().l1 + b().l2 + b().l3
          && b().updatedAt == request.time;
      }
      allow read: if true;
      allow create: if signedIn() && request.auth.uid == uid && okBa();
      allow update: if signedIn() && request.auth.uid == uid && okBa()
                    && b().l1 >= resource.data.l1
                    && b().l2 >= resource.data.l2
                    && b().l3 >= resource.data.l3;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }

    // ---- 問路小達人（4 個關卡，分數高的贏，同分比時間短） ----
    match /{col}/{uid} {
      function a() { return request.resource.data; }
      function validAskBoard() { return col.matches('^askDirections_L[1-4]$'); }
      function okAsk() {
        return a().keys().hasOnly(['name','score','time','level','updatedAt'])
          && a().name is string && a().name.size() >= 1 && a().name.size() <= 14
          && a().score is int && a().score >= 0 && a().score <= 1000
          && a().time is int && a().time >= 0 && a().time <= 99999
          && a().level is int && a().level >= 1 && a().level <= 4
          && a().updatedAt == request.time;

      allow read: if validAskBoard();
      allow create: if validAskBoard() && signedIn() && request.auth.uid == uid && okAsk();
      allow update: if validAskBoard() && signedIn() && request.auth.uid == uid && okAsk()

    match /inkdragon_boards/{theme}/entries/{uid} {
      function k() { return request.resource.data; }
      function okDragon() {

          && k().keys().hasOnly(['name','score','correct','altitude','updatedAt'])
          && k().name is string && k().name.size() >= 1 && k().name.size() <= 14
          && k().score is int && k().score >= 0 && k().score <= 100000
          && k().correct is int && k().correct >= 0 && k().correct <= 5000
          && k().altitude is int && k().altitude >= 0 && k().altitude <= 1000000
          && k().updatedAt == request.time;
      }
      allow read: if theme in ['food','fruit','animal','color','transport','body','clothes','job','weather','home'];
      allow create: if signedIn() && request.auth.uid == uid && okDragon();
      allow update: if signedIn() && request.auth.uid == uid && okDragon()
                    && k().score >= resource.data.score;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }


    // ---- 小狗找寶藏（3 關各一個榜，每台裝置每關只留一筆最好成績；2026-10-21 由 perrito 專案併進來） ----
    match /perrito_boards/{board}/entries/{uid} {
      function pd() { return request.resource.data; }
      function okPerrito() {
        return board in ['level2', 'level3', 'level4']
          && pd().keys().hasOnly(['name', 'score', 'think', 'secs', 'errors', 'presses', 'acc', 'replays', 'rank', 'updatedAt'])
          && pd().name is string && pd().name.size() > 0 && pd().name.size() <= 36
          && pd().score is int && pd().score >= 0 && pd().score <= 1000
          && pd().rank is int
          && pd().rank >= pd().score * 100000
          && pd().rank < (pd().score + 1) * 100000
          && pd().updatedAt == request.time;
      }
      allow read: if board in ['level2', 'level3', 'level4'];
      allow create: if signedIn() && request.auth.uid == uid && okPerrito();
      allow update: if signedIn() && request.auth.uid == uid && okPerrito()
                    && pd().rank > resource.data.rank;
      allow delete: if isAdmin();   // 後台排行榜總覽可刪除
    }

    // ---- 遊玩紀錄（2026-10-21 新增；每局一筆，開始時建立、玩完時補上結果；後台「遊玩紀錄」分頁用） ----
    match /plays/{playId} {
      function pl() { return request.resource.data; }
      function okPlayNew() {
        return pl().keys().hasOnly(['game', 'level', 'uid', 'name', 'day', 'startedAt', 'done'])
          && pl().game is string && pl().game.size() >= 1 && pl().game.size() <= 30
          && pl().level is string && pl().level.size() <= 30
          && pl().uid == request.auth.uid
          && pl().name is string && pl().name.size() <= 36
          && pl().day is string && pl().day.size() == 10
          && pl().startedAt == request.time
          && pl().done == false;
      }
      function okPlayEnd() {
        return resource.data.uid == request.auth.uid
          && resource.data.done == false
          && pl().diff(resource.data).affectedKeys().hasOnly(['done', 'score', 'secs', 'endedAt', 'name'])
          && pl().done == true
          && pl().endedAt == request.time
          && pl().secs is number && pl().secs >= 0 && pl().secs <= 86400
          && (!('score' in pl()) || (pl().score is number && pl().score >= -1000000000 && pl().score <= 1000000000))
          && (!('name' in pl()) || (pl().name is string && pl().name.size() <= 36));
      }
      allow create: if signedIn() && okPlayNew();
      allow update: if signedIn() && okPlayEnd();
      allow read, delete: if isAdmin();
    }

    // ---- 訪客（每台裝置、每天、每個遊戲一筆；文件編號 = 日期_遊戲_裝置編號） ----
    match /visits/{visitId} {
      function vd() { return request.resource.data; }
      allow create: if signedIn()
        && vd().keys().hasOnly(['game', 'day', 'uid', 'at'])
        && vd().game is string && vd().game.size() >= 1 && vd().game.size() <= 30
        && vd().day is string && vd().day.size() == 10
        && vd().uid == request.auth.uid
        && visitId == vd().day + '_' + vd().game + '_' + request.auth.uid
        && vd().at == request.time;
      allow update: if false;
      allow read, delete: if isAdmin();
    }
