// Memorama 排行榜的 Firebase 規則片段
// ------------------------------------------------------------
// ⚠️ 這個 Firebase 專案（memorma-24dab）也給 pop-colores-numeros 用，
//    所以「不要」把整個規則換掉！
//    請在 Firebase 後台 → Firestore Database → 規則，
//    找到 match /databases/{database}/documents { ... } 裡面，
//    把原本 memory_game_leaderboard 那一段換成下面這段（沒有的話就貼進去），
//    其他遊戲的規則保持不動，然後按「發布」。
// ------------------------------------------------------------

    match /memory_game_leaderboard/{docId} {
      // 所有人都能看排行榜
      allow read: if true;

      // 只能新增，不能修改或刪除；內容要合理才收
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['name', 'score', 'moves', 'timeSeconds', 'script', 'lang', 'createdAt'])
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 14
        && request.resource.data.moves is int
        && request.resource.data.moves >= 18
        && request.resource.data.moves <= 2000
        && request.resource.data.timeSeconds is int
        && request.resource.data.timeSeconds >= 5
        && request.resource.data.score is int
        && request.resource.data.score >= 0
        && request.resource.data.score <= 1000
        // 分數必須跟次數、時間對得上（遊戲的計分公式）
        && (
          (request.resource.data.timeSeconds < 300
            && request.resource.data.score <= 12600 / request.resource.data.moves + 301 - request.resource.data.timeSeconds)
          || (request.resource.data.timeSeconds >= 300
            && request.resource.data.score <= 12600 / request.resource.data.moves + 1)
        )
        && request.resource.data.createdAt == request.time;

      allow update, delete: if false;
    }
